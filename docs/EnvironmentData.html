EnvironmentData

Class for managing environmental sensor data from multiple APIs.

Initializes with a historical data pull and provides get_current_readings() method for fetching current data. Consolidates readings into an analytical database of parquet files that can be read with duckdb.

Commands:

  • Create HTML documentation in docs/clients: pdoc EnvironmentData.py -o docs/ --no-search
  • See experiments/1-examples-pull-data.ipynb for example usage.
  • Test with pytest tests/test_EnvironmentData.py or python experiments/run_environment_data.py

 293    def initialize_database(self, days_back: int):
 294        """
 295        Get and save historical data for each sensor.
 296
 297        Parameters
 298        ----------
 299        days_back : int
 300            Number of days of historical data to pull.
 301        """
 302
 303        # If the data already exists, initialization is not necessary.
 304        if self.cron_status != "not-initialized":
 305            return
 306
 307        # Make a log entry and gather current and starting UTC.
 308        # self.logger.info("initialize_database")
 309        current_utc = self.get_current_utc()
 310        start_utc = current_utc - days_back * 24 * 60 * 60
 311
 312        # ============ CONSERV HISTORICAL DATA PROCESSING ============
 313        conserv_readings = []
 314        if self.conserv_enabled and self.conserv_client:
 315            self.logger.info("Fetching Conserv historical data for all customers")
 316
 317            try:
 318                # Get Conserv data for the same time period as Coris
 319                conserv_data = self.conserv_client.get_historical_data(
 320                    start_utc=start_utc,
 321                    end_utc=current_utc,
 322                    test=self.testing,
 323                )
 324
 325                if conserv_data is not None and not conserv_data.is_empty():
 326                    # Conserv data is already transformed by clients/conserv_client.py
 327                    # Clean and validate Conserv data
 328                    conserv_data = self.clean_validate_sensors(
 329                        sensors=conserv_data, step="initialize_database_conserv"
 330                    )
 331
 332                    # Add to readings list
 333                    conserv_readings.append(conserv_data)
 334                    self.logger.info(
 335                        f"Successfully processed {conserv_data.shape[0]} Conserv historical records"
 336                    )
 337                else:
 338                    self.logger.info(
 339                        "No Conserv historical data available for the specified period"
 340                    )
 341
 342            except Exception as e:
 343                self.logger.warning(f"Failed to fetch Conserv historical data: {e}")
 344                # Continue without Conserv data - don't fail the entire initialization
 345
 346        # ============ CORIS HISTORICAL DATA PROCESSING ============
 347        readings = []
 348        if self.coris_enabled and self.coris_client:
 349            # Get historical data from Coris API using the client
 350            coris_readings = self.coris_client.get_historical_data_bulk(
 351                acceptable_range=self.acceptable_range,
 352                start_utc=start_utc,
 353                end_utc=current_utc,
 354                out_of_scope=self.out_of_scope,
 355                testing=self.testing,
 356                testing_sensor_ids=self.testing_sensor_ids
 357            )
 358
 359            # Clean and validate the Coris data
 360            for i, data in enumerate(coris_readings):
 361                coris_readings[i] = self.clean_validate_sensors(
 362                    sensors=data, step="initialize_database_coris"
 363                )
 364            
 365            readings.extend(coris_readings)
 366
 367        # ============ LI-COR HISTORICAL DATA PROCESSING ============
 368        licor_readings = []
 369        if self.licor_enabled and self.licor_client:
 370            # print("DEBUG: Fetching LI-COR historical data")
 371            # self.logger.info("Fetching LI-COR historical data")
 372            
 373            try:
 374                # Get LI-COR data for the same time period
 375                licor_data_list = self.licor_client.get_historical_data(
 376                    start_utc=start_utc,
 377                    end_utc=current_utc,
 378                    out_of_scope=self.out_of_scope,
 379                    testing=self.testing
 380                )
 381                
 382                if licor_data_list is not None and len(licor_data_list) > 0:
 383                    # Clean and validate each DataFrame in the list
 384                    total_records = 0
 385                    for i, data in enumerate(licor_data_list):
 386                        if data is not None and not data.is_empty():
 387                            licor_data_list[i] = self.clean_validate_sensors(
 388                                sensors=data, step="initialize_database_licor"
 389                            )
 390                            total_records += licor_data_list[i].shape[0]
 391                    
 392                    # Add all DataFrames to readings list
 393                    licor_readings.extend(licor_data_list)
 394                    # print(f"DEBUG: Successfully processed {total_records} LI-COR historical records from {len(licor_data_list)} DataFrames")
 395                    # self.logger.info(
 396                    #     f"Successfully processed {total_records} LI-COR historical records from {len(licor_data_list)} DataFrames"
 397                    # )
 398                else:
 399                    print("DEBUG: No LI-COR historical data available for the specified period")
 400                    self.logger.info(
 401                        "No LI-COR historical data available for the specified period"
 402                    )
 403                    
 404            except Exception as e:
 405                print(f"DEBUG: Failed to fetch LI-COR historical data: {e}")
 406                self.logger.warning(f"Failed to fetch LI-COR historical data: {e}")
 407                # Continue without LI-COR data - don't fail the entire initialization
 408        else:
 409            print(f"DEBUG: LI-COR not enabled or client not initialized. Enabled: {self.licor_enabled}, Client: {self.licor_client}")
 410
 411        # ============ COMBINE ALL DATA SOURCES ============
 412        # Combine Conserv, Coris, and LI-COR readings into a single polars DataFrame
 413        all_readings = conserv_readings + readings + licor_readings
 414        dt = (
 415            polars.concat(all_readings, how="diagonal")
 416            if all_readings
 417            else polars.DataFrame()
 418        )
 419
 420        if dt.is_empty():
 421            error_msg = "No data from any source - initialization failed"
 422            self.logger.error(error_msg)
 423            raise RuntimeError(error_msg)
 424
 425        # Standardize the DataFrame (rename columns, filter, and reorder)
 426        dt = self.standardize_sensor_dataframe(dt)
 427
 428        # Write the database file.
 429        dt.write_parquet(f"{self.data_path}/sensor_readings.parquet")
 430        self.logger.info(
 431            f"Historical data initialization complete: {dt.shape[0]} total records from {len(all_readings)} sources"
 432        )
 433
 434        self.update_cron_status("initialized")

 566        # )

 664    def consolidate_readings(self):
 665        """
 666        Combine new and historical readings into one database. Build (or re-build) the analytical tables.
 667        If successful, remove the new-readings files that have been processed into the database.
 668        Meant to run as a daily batch process to consolidate readings made throughout the day.
 669        """
 670
 671        if self.cron_status == "not-initialized":
 672            return
 673
 674        # Make a log entry.
 675        # self.logger.info("consolidate_readings")
 676        new_readings = []
 677
 678        # Read new readings from the parquet files saved by calls to get_current_readings.
 679        # These are deleted after each consolidation, so these files will always be the un-consolidated files.
 680        if os.path.exists(f"{self.data_path}/new-readings/"):
 681            files = os.listdir(f"{self.data_path}/new-readings/")
 682            files_read = []
 683            for file in files:
 684
 685                # Read any per-file parquet and collect them
 686                try:
 687                    new_readings.append(
 688                        polars.read_parquet(f"{self.data_path}/new-readings/{file}")
 689                    )
 690                    files_read.append(file)
 691                except Exception:
 692                    # swallow and continue to the next file
 693                    pass
 694        else:
 695            files_read = []
 696
 697        # Combine the readings into a single polars DataFrame.
 698        dt = polars.concat(new_readings) if new_readings else polars.DataFrame()
 699        self.logger.info(f"{dt.shape[0]} new readings after schema enforcement.")
 700
 701        # Clean the data.
 702        try:
 703            historical = polars.read_parquet(
 704                f"{self.data_path}/sensor_readings.parquet"
 705            )
 706        except FileNotFoundError:
 707            # Create empty DataFrame with correct schema for first run
 708            historical = polars.DataFrame()
 709
 710        # Only clean/validate new readings if there are any
 711        if not dt.is_empty():
 712            dt = self.clean_validate_sensors(
 713                sensors=dt, historical=historical, step="consolidate_readings"
 714            )
 715
 716            # Apply schema gate before concatenating with historical data
 717            dt = self.enforce_schema(dt, "CleanedNewReadings")
 718            if not historical.is_empty():
 719                historical = self.enforce_schema(historical, "HistoricalData")
 720
 721            # Append these to the database.
 722            dt = polars.concat([historical, dt], how="diagonal")
 723        else:
 724            # No new readings - just use historical data
 725            self.logger.info("No new readings to consolidate. Using historical data only.")
 726            dt = historical
 727
 728        # Remove duplicate readings based on SensorID and SensorReadingUTC
 729        # This handles cases where Conserv (15-min intervals) might pull the same readings multiple times
 730        initial_count = dt.shape[0]
 731        dt = dt.unique(subset=["SensorID", "SensorReadingUTC"])
 732        final_count = dt.shape[0]
 733        
 734        if initial_count != final_count:
 735            duplicates_removed = initial_count - final_count
 736            self.logger.info(f"Removed {duplicates_removed} duplicate readings (SensorID + SensorReadingUTC)")
 737
 738        # Add time difference between readings for each sensor within each source, device, and type
 739        # Only calculate for non-Historical data to avoid mixing backfilled data with real-time readings
 740        if not dt.is_empty():
 741            dt = dt.sort(["Source", "DeviceID", "SensorType", "SensorID", "SensorReadingUTC"])
 742            
 743            # Calculate time differences grouping by Historical flag to prevent mixing historical and real-time data
 744            dt = dt.with_columns(
 745                (
 746                    polars.col("SensorReadingUTC")
 747                    - polars.col("SensorReadingUTC").shift(1).over(["Source", "DeviceID", "SensorType", "SensorID", "Historical"])
 748                ).alias("SensorReadingUTC_SecondsFromPrior")
 749            )
 750            
 751            # Then set Historical data timing to None since it shouldn't have real-time timing
 752            dt = dt.with_columns(
 753                polars.when(polars.col("Historical") == True)
 754                .then(None)
 755                .otherwise(polars.col("SensorReadingUTC_SecondsFromPrior"))
 756                .alias("SensorReadingUTC_SecondsFromPrior")
 757            )
 758        else:
 759            dt = dt.with_columns(
 760                polars.lit(None, dtype=polars.Int64).alias(
 761                    "SensorReadingUTC_SecondsFromPrior"
 762                )
 763            )
 764
 765        # Standardize the DataFrame (rename columns, filter, and reorder)
 766        dt = self.standardize_sensor_dataframe(dt)
 767
 768        # Write the file.
 769        dt.write_parquet(f"{self.data_path}/sensor_readings.parquet")
 770        # self.logger.info(f"{dt.shape[0]} total readings.")
 771
 772        # Run validation with return_results=True to collect all validation results
 773        validation_results = self.validate_sensors(
 774            sensors=dt,
 775            step="consolidate_final",
 776            return_results=True
 777        )
 778
 779        # If all this was successful, remove the new-readings files to prepare for the next consolidation.
 780        for file in files_read:
 781            os.remove(f"{self.data_path}/new-readings/{file}")
 782
 783        # Refresh the devices table.
 784        devices = self.build_devices(dt)
 785        self.validate_devices(devices)
 786        devices.write_parquet(f"{self.data_path}/device_readings.parquet")
 787
 788        # Update lookup tables and cubes.
 789        self.update_lookups()
 790        self.update_cubes()
 791        
 792        # Generate diagnostics CSV including gaps and alerts
 793        # This must happen AFTER update_lookups() so sensors.parquet exists for building info check
 794        validation.generate_validation_results(
 795            sensors=dt,
 796            validation_results=validation_results,
 797            acceptable_range=self.acceptable_range,
 798            data_path=self.data_path,
 799            logger=self.logger,
 800            step="consolidate_readings"
 801        )
 802        
 803        # Export data to Excel.
 804        self.export_to_excel()

 983        """
 984        Send alerts based on the sensor data.
 985        Currently, this function saves alerts to "alerts.txt" in the data folder.
 986        Later, this can be connected to an messaging system like Twilio.
 987
 988        Parameters
 989        ----------
 990        sensors: polars.DataFrame

 993        utc: int
 994            Current UTC timestamp. Passed from another function to keep alerts aligned with QueryUTC.
 995        """
 996
 997        alerts = []
 998        for reading in self.acceptable_range:
 999
1000            if len(self.acceptable_range[reading]) == 0:
1001                continue
1002
1003            reading_idx = self.which(sensors.columns == reading)
1004
1005            for i, row in sensors.iter_rows():
1006                if (not not row[reading_idx].is_null()) and (
1007                    row[reading_idx] < self.acceptable_range[reading][0]
1008                    or row[reading] > self.acceptable_range[reading][1]
1009                ):
1010                    alerts.append(row)
1011
1012        # Write the alerts to a file.
1013        # Later, this can be connected to an alerting system like Twilio.
1014        if len(alerts) > 0:
1015            with open(f"{self.data_path}/alerts.txt", "w") as f:
1016                f.write("\n".join(alerts))

 294    def initialize_database(self, days_back: int):
 295        """
 296        Get and save historical data for each sensor.
 297
 298        Parameters
 299        ----------
 300        days_back : int
 301            Number of days of historical data to pull.
 302        """
 303
 304        # If the data already exists, initialization is not necessary.
 305        if self.cron_status != "not-initialized":
 306            return
 307
 308        # Make a log entry and gather current and starting UTC.
 309        # self.logger.info("initialize_database")
 310        current_utc = self.get_current_utc()
 311        start_utc = current_utc - days_back * 24 * 60 * 60
 312
 313        # ============ CONSERV HISTORICAL DATA PROCESSING ============
 314        conserv_readings = []
 315        if self.conserv_enabled and self.conserv_client:
 316            self.logger.info("Fetching Conserv historical data for all customers")
 317
 318            try:
 319                # Get Conserv data for the same time period as Coris
 320                conserv_data = self.conserv_client.get_historical_data(
 321                    start_utc=start_utc,
 322                    end_utc=current_utc,
 323                    test=self.testing,
 324                )
 325
 326                if conserv_data is not None and not conserv_data.is_empty():
 327                    # Conserv data is already transformed by clients/conserv_client.py
 328                    # Clean and validate Conserv data
 329                    conserv_data = self.clean_validate_sensors(
 330                        sensors=conserv_data, step="initialize_database_conserv"
 331                    )
 332
 333                    # Add to readings list
 334                    conserv_readings.append(conserv_data)
 335                    self.logger.info(
 336                        f"Successfully processed {conserv_data.shape[0]} Conserv historical records"
 337                    )
 338                else:
 339                    self.logger.info(
 340                        "No Conserv historical data available for the specified period"
 341                    )
 342
 343            except Exception as e:
 344                self.logger.warning(f"Failed to fetch Conserv historical data: {e}")
 345                # Continue without Conserv data - don't fail the entire initialization
 346
 347        # ============ CORIS HISTORICAL DATA PROCESSING ============
 348        readings = []
 349        if self.coris_enabled and self.coris_client:
 350            # Get historical data from Coris API using the client
 351            coris_readings = self.coris_client.get_historical_data_bulk(
 352                acceptable_range=self.acceptable_range,
 353                start_utc=start_utc,
 354                end_utc=current_utc,
 355                out_of_scope=self.out_of_scope,
 356                testing=self.testing,
 357                testing_sensor_ids=self.testing_sensor_ids
 358            )
 359
 360            # Clean and validate the Coris data
 361            for i, data in enumerate(coris_readings):
 362                coris_readings[i] = self.clean_validate_sensors(
 363                    sensors=data, step="initialize_database_coris"
 364                )
 365            
 366            readings.extend(coris_readings)
 367
 368        # ============ LI-COR HISTORICAL DATA PROCESSING ============
 369        licor_readings = []
 370        if self.licor_enabled and self.licor_client:
 371            # print("DEBUG: Fetching LI-COR historical data")
 372            # self.logger.info("Fetching LI-COR historical data")
 373            
 374            try:
 375                # Get LI-COR data for the same time period
 376                licor_data_list = self.licor_client.get_historical_data(
 377                    start_utc=start_utc,
 378                    end_utc=current_utc,
 379                    out_of_scope=self.out_of_scope,
 380                    testing=self.testing
 381                )
 382                
 383                if licor_data_list is not None and len(licor_data_list) > 0:
 384                    # Clean and validate each DataFrame in the list
 385                    total_records = 0
 386                    for i, data in enumerate(licor_data_list):
 387                        if data is not None and not data.is_empty():
 388                            licor_data_list[i] = self.clean_validate_sensors(
 389                                sensors=data, step="initialize_database_licor"
 390                            )
 391                            total_records += licor_data_list[i].shape[0]
 392                    
 393                    # Add all DataFrames to readings list
 394                    licor_readings.extend(licor_data_list)
 395                    # print(f"DEBUG: Successfully processed {total_records} LI-COR historical records from {len(licor_data_list)} DataFrames")
 396                    # self.logger.info(
 397                    #     f"Successfully processed {total_records} LI-COR historical records from {len(licor_data_list)} DataFrames"
 398                    # )
 399                else:
 400                    print("DEBUG: No LI-COR historical data available for the specified period")
 401                    self.logger.info(
 402                        "No LI-COR historical data available for the specified period"
 403                    )
 404                    
 405            except Exception as e:
 406                print(f"DEBUG: Failed to fetch LI-COR historical data: {e}")
 407                self.logger.warning(f"Failed to fetch LI-COR historical data: {e}")
 408                # Continue without LI-COR data - don't fail the entire initialization
 409        else:
 410            print(f"DEBUG: LI-COR not enabled or client not initialized. Enabled: {self.licor_enabled}, Client: {self.licor_client}")
 411
 412        # ============ COMBINE ALL DATA SOURCES ============
 413        # Combine Conserv, Coris, and LI-COR readings into a single polars DataFrame
 414        all_readings = conserv_readings + readings + licor_readings
 415        dt = (
 416            polars.concat(all_readings, how="diagonal")
 417            if all_readings
 418            else polars.DataFrame()
 419        )
 420
 421        if dt.is_empty():
 422            error_msg = "No data from any source - initialization failed"
 423            self.logger.error(error_msg)
 424            raise RuntimeError(error_msg)
 425
 426        # Standardize the DataFrame (rename columns, filter, and reorder)
 427        dt = self.standardize_sensor_dataframe(dt)
 428
 429        # Write the database file.
 430        dt.write_parquet(f"{self.data_path}/sensor_readings.parquet")
 431        self.logger.info(
 432            f"Historical data initialization complete: {dt.shape[0]} total records from {len(all_readings)} sources"
 433        )
 434
 435        self.update_cron_status("initialized")

 665    def consolidate_readings(self):
 666        """
 667        Combine new and historical readings into one database. Build (or re-build) the analytical tables.
 668        If successful, remove the new-readings files that have been processed into the database.
 669        Meant to run as a daily batch process to consolidate readings made throughout the day.
 670        """
 671
 672        if self.cron_status == "not-initialized":
 673            return
 674
 675        # Make a log entry.
 676        # self.logger.info("consolidate_readings")
 677        new_readings = []
 678
 679        # Read new readings from the parquet files saved by calls to get_current_readings.
 680        # These are deleted after each consolidation, so these files will always be the un-consolidated files.
 681        if os.path.exists(f"{self.data_path}/new-readings/"):
 682            files = os.listdir(f"{self.data_path}/new-readings/")
 683            files_read = []
 684            for file in files:
 685
 686                # Read any per-file parquet and collect them
 687                try:
 688                    new_readings.append(
 689                        polars.read_parquet(f"{self.data_path}/new-readings/{file}")
 690                    )
 691                    files_read.append(file)
 692                except Exception:
 693                    # swallow and continue to the next file
 694                    pass
 695        else:
 696            files_read = []
 697
 698        # Combine the readings into a single polars DataFrame.
 699        dt = polars.concat(new_readings) if new_readings else polars.DataFrame()
 700        self.logger.info(f"{dt.shape[0]} new readings after schema enforcement.")
 701
 702        # Clean the data.
 703        try:
 704            historical = polars.read_parquet(
 705                f"{self.data_path}/sensor_readings.parquet"
 706            )
 707        except FileNotFoundError:
 708            # Create empty DataFrame with correct schema for first run
 709            historical = polars.DataFrame()
 710
 711        # Only clean/validate new readings if there are any
 712        if not dt.is_empty():
 713            dt = self.clean_validate_sensors(
 714                sensors=dt, historical=historical, step="consolidate_readings"
 715            )
 716
 717            # Apply schema gate before concatenating with historical data
 718            dt = self.enforce_schema(dt, "CleanedNewReadings")
 719            if not historical.is_empty():
 720                historical = self.enforce_schema(historical, "HistoricalData")
 721
 722            # Append these to the database.
 723            dt = polars.concat([historical, dt], how="diagonal")
 724        else:
 725            # No new readings - just use historical data
 726            self.logger.info("No new readings to consolidate. Using historical data only.")
 727            dt = historical
 728
 729        # Remove duplicate readings based on SensorID and SensorReadingUTC
 730        # This handles cases where Conserv (15-min intervals) might pull the same readings multiple times
 731        initial_count = dt.shape[0]
 732        dt = dt.unique(subset=["SensorID", "SensorReadingUTC"])
 733        final_count = dt.shape[0]
 734        
 735        if initial_count != final_count:
 736            duplicates_removed = initial_count - final_count
 737            self.logger.info(f"Removed {duplicates_removed} duplicate readings (SensorID + SensorReadingUTC)")
 738
 739        # Add time difference between readings for each sensor within each source, device, and type
 740        # Only calculate for non-Historical data to avoid mixing backfilled data with real-time readings
 741        if not dt.is_empty():
 742            dt = dt.sort(["Source", "DeviceID", "SensorType", "SensorID", "SensorReadingUTC"])
 743            
 744            # Calculate time differences grouping by Historical flag to prevent mixing historical and real-time data
 745            dt = dt.with_columns(
 746                (
 747                    polars.col("SensorReadingUTC")
 748                    - polars.col("SensorReadingUTC").shift(1).over(["Source", "DeviceID", "SensorType", "SensorID", "Historical"])
 749                ).alias("SensorReadingUTC_SecondsFromPrior")
 750            )
 751            
 752            # Then set Historical data timing to None since it shouldn't have real-time timing
 753            dt = dt.with_columns(
 754                polars.when(polars.col("Historical") == True)
 755                .then(None)
 756                .otherwise(polars.col("SensorReadingUTC_SecondsFromPrior"))
 757                .alias("SensorReadingUTC_SecondsFromPrior")
 758            )
 759        else:
 760            dt = dt.with_columns(
 761                polars.lit(None, dtype=polars.Int64).alias(
 762                    "SensorReadingUTC_SecondsFromPrior"
 763                )
 764            )
 765
 766        # Standardize the DataFrame (rename columns, filter, and reorder)
 767        dt = self.standardize_sensor_dataframe(dt)
 768
 769        # Write the file.
 770        dt.write_parquet(f"{self.data_path}/sensor_readings.parquet")
 771        # self.logger.info(f"{dt.shape[0]} total readings.")
 772
 773        # Run validation with return_results=True to collect all validation results
 774        validation_results = self.validate_sensors(
 775            sensors=dt,
 776            step="consolidate_final",
 777            return_results=True
 778        )
 779
 780        # If all this was successful, remove the new-readings files to prepare for the next consolidation.
 781        for file in files_read:
 782            os.remove(f"{self.data_path}/new-readings/{file}")
 783
 784        # Refresh the devices table.
 785        devices = self.build_devices(dt)
 786        self.validate_devices(devices)
 787        devices.write_parquet(f"{self.data_path}/device_readings.parquet")
 788
 789        # Update lookup tables and cubes.
 790        self.update_lookups()
 791        self.update_cubes()
 792        
 793        # Generate diagnostics CSV including gaps and alerts
 794        # This must happen AFTER update_lookups() so sensors.parquet exists for building info check
 795        validation.generate_validation_results(
 796            sensors=dt,
 797            validation_results=validation_results,
 798            acceptable_range=self.acceptable_range,
 799            data_path=self.data_path,
 800            logger=self.logger,
 801            step="consolidate_readings"
 802        )
 803        
 804        # Export data to Excel.
 805        self.export_to_excel()

 807    def match_types(
 808        self, data: polars.DataFrame, match: polars.DataFrame
 809    ) -> polars.DataFrame:
 810        """
 811        Modify the column data types of one DataFrame to match another.
 812        Useful to prevent errors during concatenation.
 813
 814        Parameters
 815        ----------
 816        data : polars.DataFrame
 817            DataFrame to match (data types may get changed).
 818
 819        match : polars.DataFrame
 820            DataFrame to match column data types to.
 821
 822        Returns
 823        -------
 824        polars.DataFrame: Same data as the "data" DataFrame, but with types that match the "match" DataFrame.
 825        """
 826
 827        for col in match.columns:
 828            if col in data.columns and data[col].dtype != match[col].dtype:
 829                data = data.with_columns(data[col].cast(match[col].dtype))
 830
 831        return data
